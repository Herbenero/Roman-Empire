function processHit(num) { if (state.gameEnded) return; pushHistory(); const player = getCurrentPlayer(); const multiplier = state.activeMultiplier; const seg = state.board[num]; let message = ""; if (!seg.owner) { // Unclaimed segment if (multiplier >= 2) { seg.owner = player.id; seg.shield = true; message = `${player.name} hit ${formatHit(multiplier, num)} and captured ${num} WITH a shield!`; } else { seg.owner = player.id; seg.shield = false; message = `${player.name} hit Single ${num} and captured segment ${num}!`; } } else if (seg.owner === player.id) { // Player's own segment if (!seg.shield) { seg.shield = true; message = `${player.name} hit ${formatHit(multiplier, num)} on segment ${num} and added a shield!`; } else { message = `${player.name} hit ${formatHit(multiplier, num)} on segment ${num}, but it was already shielded.`; } } else { // Opponent's segment const defenderName = state.players.find((p) => p.id === seg.owner)?.name || "Enemy"; if (seg.shield) { if (multiplier >= 3) { // Multiplier of 3+ (Triple/Quad) bypasses shield and frees the segment seg.owner = null; seg.shield = false; message = `${player.name} hit ${formatHit(multiplier, num)}! Bypassed shield and freed ${defenderName}'s segment!`; } else if (multiplier === 2) { // Double strips the shield, segment remains with opponent seg.shield = false; message = `${player.name} hit Double ${num} and stripped ${defenderName}'s shield!`; } else { // Single is repelled message = `${player.name} hit Single ${num} against ${defenderName}'s shielded segment. Repelled!`; } } else { // Unshielded opponent segment is freed (unclaimed) rather than captured seg.owner = null; seg.shield = false; message = `${player.name} hit ${formatHit(multiplier, num)} and freed ${defenderName}'s segment!`; } } // Reset multiplier back to 1 after dart is processed state.activeMultiplier = 1; state.dartsThisTurn += 1; state.dartsThrown += 1; // Add message to log state.log.unshift(`[R${state.currentRound} D${state.dartsThisTurn}] ${message}`); // Check turn progression / round limits if (state.dartsThisTurn >= 3) { advanceTurn(); } else { renderAll(); } }
